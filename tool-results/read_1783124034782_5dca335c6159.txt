   604→          </div>

   609→          </div>
   610→        </div>
   611→
   612→        <DialogFooter className="gap-2">
   613→          {productId && (
   614→            <Button
   615→              variant="destructive"
   616→              onClick={async () => {
   617→                if (confirm(t('inv.deleteConfirm'))) {
   618→                  await fetch(`/api/products/${productId}`, { method: 'DELETE' })
   619→                  toast.success('Product deleted')
   620→                  triggerRefresh()
   621→                  onOpenChange(false)
   622→                  setEditingProductId(null)
   623→                }
   624→              }}
   625→              className="h-11"
   626→            >
   627→              {t('common.delete')}
   628→            </Button>
   629→          )}
   630→          <Button variant="outline" onClick={() => onOpenChange(false)} className="h-11">
   631→            {t('common.cancel')}
   632→          </Button>
   633→          <Button onClick={handleSave} disabled={saving} className="h-11 flex-1">
   634→            {saving ? 'Saving…' : t('common.save')}
   635→          </Button>
   636→        </DialogFooter>
   637→      </DialogContent>
   638→    </Dialog>
   639→  )
   640→}
