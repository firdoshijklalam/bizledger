   100→function gradientFor(seed: string): string {
   101→  let hash = 0
   102→  for (let i = 0; i < seed.length; i++) hash = (hash * 31 + seed.charCodeAt(i)) | 0
   103→  return AVATAR_GRADIENTS[Math.abs(hash) % AVATAR_GRADIENTS.length]
   104→}

   820→        )}

   826→// ============================ Product Card ============================
   827→function ProductCard({
   828→  product,
   829→  currency,
   830→  onAdd,
   831→}: {
   832→  product: StoreProduct
   833→  currency: string
   834→  onAdd: (quantity: number, unitPrice: number, unitLabel: string) => void
   835→}) {
   836→  const [selectedWeight, setSelectedWeight] = useState(1)
   837→  const [useRetail, setUseRetail] = useState(false)
   838→
   839→  const discount =
   840→    product.mrp && product.mrp > product.salePrice
   841→      ? Math.round(((product.mrp - product.salePrice) / product.mrp) * 100)
   842→      : 0
   843→  const lowStock = product.stock > 0 && product.stock < LOW_STOCK_THRESHOLD
   844→  const outOfStock = product.stock <= 0
   845→
   846→  // PRD Part 35 §3.2: Dynamic weight options for loose/retail products
   847→  const weightOptions = product.retailEnabled && product.retailUnit
   848→    ? [1, 2, 5, 10]
   849→    : []
   850→
   851→  const currentPrice = useRetail && product.retailSalePrice
   852→    ? product.retailSalePrice * selectedWeight
   853→    : product.salePrice * selectedWeight
   854→
   855→  const currentUnitLabel = useRetail && product.retailUnit
   856→    ? `${selectedWeight} ${product.retailUnit}`
   857→    : selectedWeight > 1
   858→    ? `${selectedWeight} ${product.unit}s`
   859→    : product.unit || ''
   860→
   861→  const handleAdd = () => {
   862→    onAdd(selectedWeight, useRetail && product.retailSalePrice ? product.retailSalePrice : product.salePrice, currentUnitLabel)
   863→  }
   864→
   865→  return (
   866→    <motion.div
   867→      initial={{ opacity: 0, y: 8 }}
   868→      animate={{ opacity: 1, y: 0 }}
   869→      whileHover={{ scale: 1.02, boxShadow: '0 10px 30px -10px rgba(0,0,0,0.15)' }}
   870→      transition={{ type: 'spring', stiffness: 300, damping: 20 }}
   871→      className="bg-card/80 backdrop-blur-xl border border-border rounded-2xl overflow-hidden flex flex-col"
   872→    >
   873→      {/* Image placeholder with first letter */}
   874→      <div
   875→        className={`relative aspect-square bg-gradient-to-br ${gradientFor(
   876→          product.id
   877→        )} flex items-center justify-center`}
   878→      >
   879→        <span className="text-4xl font-bold text-white/90 drop-shadow-sm">
   880→          {product.name.charAt(0).toUpperCase()}
   881→        </span>
   882→        {/* Discount badge */}
   883→        {discount > 0 && (
   884→          <span className="absolute top-2 left-2 px-1.5 py-0.5 rounded-md bg-rose-500 text-white text-[10px] font-bold">
   885→            {discount}% OFF
   886→          </span>
   887→        )}
   888→        {/* Loose product badge */}
   889→        {product.retailEnabled && (
   890→          <span className="absolute top-2 left-2 px-1.5 py-0.5 rounded-md bg-violet-500 text-white text-[10px] font-bold">
   891→            Loose
   892→          </span>
   893→        )}
   894→        {/* Stock badge */}
   895→        <div className="absolute top-2 right-2">
   896→          {outOfStock ? (
   897→            <span className="px-1.5 py-0.5 rounded-md bg-zinc-900/70 text-white text-[10px] font-medium">
   898→              Out of stock
   899→            </span>
   900→          ) : lowStock ? (
   901→            <span className="px-1.5 py-0.5 rounded-md bg-amber-500/90 text-white text-[10px] font-medium">
   902→              Low stock
   903→            </span>
   904→          ) : (
   905→            <span className="px-1.5 py-0.5 rounded-md bg-emerald-500/90 text-white text-[10px] font-medium">
   906→              In stock
   907→            </span>
   908→          )}
   909→        </div>
   910→      </div>
   911→
   912→      {/* Body */}
   913→      <div className="p-2.5 flex flex-col flex-1 gap-1.5">
   914→        {/* Category */}
   915→        {product.category && (
   916→          <div className="flex items-center gap-1 flex-wrap">
   917→            <Badge
   918→              variant="secondary"
   919→              className="bg-muted text-muted-foreground text-[9px] px-1.5 py-0 h-4"
   920→            >
   921→              {product.category}
   922→            </Badge>
   923→            {product.subCategory && (
   924→              <span className="text-[9px] text-muted-foreground/70 truncate">
   925→                · {product.subCategory}
   926→              </span>
   927→            )}
   928→          </div>
   929→        )}
   930→
   931→        {/* Name */}
   932→        <h3 className="text-sm font-semibold leading-snug line-clamp-2 min-h-[2.4em]">
   933→          {product.name}
   934→        </h3>
   935→
   936→        {/* PRD Part 35 §1.2: AI-generated description */}
   937→        {(product as any).description && (
   938→          <p className="text-[10px] text-muted-foreground line-clamp-2">
   939→            {(product as any).description}
   940→          </p>
   941→        )}
   942→
   943→        {/* Price */}
   944→        <div className="mt-auto">
   945→          <div className="flex items-baseline gap-1.5">
   946→            <span className="text-base font-bold text-emerald-600 dark:text-emerald-400 tabular">
   947→              {formatCurrency(product.salePrice, currency)}
   948→            </span>
   949→            {product.unit && (
   950→              <span className="text-[10px] text-muted-foreground">/ {product.unit}</span>

   952→          </div>
   953→          {product.mrp && product.mrp > product.salePrice && (

   955→              {formatCurrency(product.mrp, currency)}

   958→          {/* Retail price alternative */}

  1001→          </div>
  1002→        )}
  1003→
  1004→        {/* Dynamic total price preview */}
  1005→        {selectedWeight > 1 && (

  1010→
  1011→        {/* Add to Cart */}
  1012→        <Button
  1013→          onClick={handleAdd}
  1014→          disabled={outOfStock}
  1015→          size="sm"
  1016→          className="w-full mt-1 h-8 rounded-lg bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-semibold disabled:opacity-40 disabled:cursor-not-allowed"
  1017→        >
  1018→          <Plus className="w-3.5 h-3.5" />

  1021→      </div>
  1022→    </motion.div>
  1023→  )
  1024→}
  1025→
