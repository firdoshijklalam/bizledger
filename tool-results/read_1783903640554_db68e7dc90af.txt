    33→interface PartyDetailData extends Party {
    34→  transactions: Transaction[]
    35→  invoices: Invoice[]

    38→
    39→export function PartyDetail({ partyId }: { partyId: string }) {
    40→  const { setSelectedPartyId, setActiveView, setShowInvoiceForm, business, setSelectedInvoiceId, setEditingPartyId, editingPartyId, returnToView, setReturnToView } = useAppStore()
    41→  const { t } = useI18n()
    42→  const { data, loading, error, refetch } = useFetch<PartyDetailData>(`/api/parties/${partyId}`, [partyId])
    43→  const [showTxn, setShowTxn] = useState(false)
    44→  const [showSettle, setShowSettle] = useState(false)
    45→  const [showNote, setShowNote] = useState(false)
    46→  const [showCompare, setShowCompare] = useState(false)
    47→  const [compareProduct, setCompareProduct] = useState<string>('')
    48→  // PRD Part 32 §4: family/partner manager modals + fingerprint registration
    49→  const [showFamily, setShowFamily] = useState(false)
    50→  const [showPartner, setShowPartner] = useState(false)
    51→  const [showFingerprintRegister, setShowFingerprintRegister] = useState(false)
    52→
    53→  // PRD Part 7 §3: restore scroll on back button (party detail → khata)
    54→  const { restore: restoreScrollPos } = useScrollStore()
    55→  const handleBack = () => {
    56→    if (returnToView) {
    57→      setActiveView(returnToView)
    58→      setReturnToView(null)
    59→    } else {
    60→      setActiveView('khata')
    61→    }
    62→    setSelectedPartyId(null)
    63→    // Restore scroll after view transition
    64→    setTimeout(() => restoreScrollPos('khata'), 50)
    65→    setTimeout(() => restoreScrollPos('khata'), 150)
    66→    setTimeout(() => restoreScrollPos('khata'), 300)
    67→  }
    68→
    69→  // PRD Part 24: Source Products section for suppliers (fetches catalog)
    70→  const isSupplier = data?.type === 'supplier' || data?.type === 'both'
    71→  const isCustomer = data?.type === 'customer' || data?.type === 'both'
    72→  const { data: supplierCatalog } = useFetch<SupplierCatalogItem[]>(
    73→    isSupplier ? `/api/suppliers/${partyId}/catalog` : null,
    74→    [partyId, isSupplier]
    75→  )
    76→  // PRD Part 32 §3.1: check shared defaulter registry for this party's phone
    77→  const { data: defaulterMatch } = useFetch<any[]>(
    78→    data?.phone ? `/api/defaulter-registry?phone=${encodeURIComponent(data.phone)}` : null,
    79→    [data?.phone]
    80→  )
    81→  const activeDefaulter = defaulterMatch && defaulterMatch.length > 0 ? defaulterMatch[0] : null
    82→  // Multi-select state (PRD Part 7 §4)
    83→  const [multiSelectMode, setMultiSelectMode] = useState(false)
    84→  const [selectedTxIds, setSelectedTxIds] = useState<Set<string>>(new Set())
    85→  // Share sheet state (PRD Part 10 §2)
    86→  const [showShareSheet, setShowShareSheet] = useState(false)
    87→  const [shareSheetText, setShareSheetText] = useState('')
    88→  const [shareSheetTitle, setShareSheetTitle] = useState('')
    89→
    90→  // Multi-select handlers (PRD Part 7 §4) — must be before early return
    91→  const toggleTxSelection = useCallback((txId: string) => {
    92→    setSelectedTxIds((prev) => {
    93→      const next = new Set(prev)
    94→      if (next.has(txId)) next.delete(txId)
    95→      else next.add(txId)
    96→      return next
    97→    })
    98→  }, [])
    99→
   100→  const selectAllTx = () => {
   101→    if (!data) return
   102→    setSelectedTxIds(new Set(data.transactions.map((t) => t.id)))
   103→  }
   104→
   105→  const deselectAllTx = () => {
   106→    setSelectedTxIds(new Set())
   107→    setMultiSelectMode(false)
   108→  }
   109→
   110→  const handleShareSelected = () => {
   111→    if (!data) return
   112→    const selected = data.transactions.filter((t) => selectedTxIds.has(t.id))
   113→    if (selected.length === 0) return
   114→    // PRD Part 10: Open dynamic share sheet instead of direct WhatsApp
   115→    const lines = [
   116→      `${business?.name || 'BizLedger'} — Selected Transactions`,
   117→      `Customer: ${data.name}`,
   118→      ``,
   119→    ]
   120→    selected.forEach((tx) => {
   121→      const isCredit = tx.type === 'credit'
   122→      lines.push(`${formatDate(tx.createdAt)} | ${isCredit ? 'পেলাম' : 'দিলাম'} | ${formatCurrency(tx.amount, currency)} | ${tx.description || ''}`)
   123→    })
   124→    lines.push(``, `Total: ${selected.length} transactions`)
   125→    setShareSheetText(lines.join('\n'))
   126→    setShareSheetTitle(`${selected.length} Transactions Statement`)
   127→    setShowShareSheet(true)
   128→  }
   129→
   130→  const handleShareStatement = () => {
   131→    if (!data) return
   132→    // PRD Part 10: Share full statement via dynamic share sheet
   133→    const lines = [
   134→      `${business?.name || 'BizLedger'} — Ledger Statement`,
   135→      `Customer: ${data.name}`,
   136→      ``,
   137→      `Date | Type | Amount | Description`,
   138→    ]
   139→    data.transactions.forEach((tx) => {
   140→      const isCredit = tx.type === 'credit'
   141→      lines.push(`${formatDate(tx.createdAt)} | ${isCredit ? 'পেলাম' : 'দিলাম'} | ${formatCurrency(tx.amount, currency)} | ${tx.description || ''}`)
   142→    })
   143→    lines.push(``, `Current Balance: ${formatCurrency(data.balance, currency)}`)
   144→    setShareSheetText(lines.join('\n'))
   145→    setShareSheetTitle('Full Ledger Statement')
   146→    setShowShareSheet(true)
   147→  }
   148→
   149→  // §4: Loading + error states — prevent blank screen
   150→  if (loading) {
   151→    return (
   152→      <div className="space-y-4 p-4">
   153→        <button onClick={handleBack} className="flex items-center gap-2 text-sm text-muted-foreground">
   154→          <ArrowLeft className="w-4 h-4" /> পিছনে
   155→        </button>
   156→        <div className="flex items-center justify-center py-20">
   157→          <Loader2 className="w-8 h-8 animate-spin text-muted-foreground" />
   158→        </div>
   159→      </div>
   160→    )
   161→  }
   162→  if (error || !data) {
   163→    return (
   164→      <div className="space-y-4 p-4">
   165→        <button onClick={handleBack} className="flex items-center gap-2 text-sm text-muted-foreground">
   166→          <ArrowLeft className="w-4 h-4" /> পিছনে
   167→        </button>
   168→        <div className="flex flex-col items-center justify-center py-20 gap-3">
   169→          <AlertTriangle className="w-10 h-10 text-muted-foreground/50" />
   170→          <p className="text-sm text-muted-foreground">ডেটা লোড করা যায়নি</p>
   171→          <button onClick={() => refetch()} className="px-4 py-2 rounded-xl bg-primary text-primary-foreground text-sm font-medium">
   172→            আবার চেষ্টা করুন
   173→          </button>
   174→        </div>
   175→      </div>
   176→    )
   177→  }
   178→  const currency = business?.currency || 'INR'
   179→  const meta = GRADE_META[data.qualityGrade]
   180→  const isReceivable = data.balance > 0
   181→  const isPayable = data.balance < 0
   182→
   183→  const handleSettle = async (amount: number) => {
   184→    await apiPost('/api/transactions', {
   185→      partyId: data.id,
   186→      type: isReceivable ? 'credit' : 'debit',
   187→      amount,
   188→      description: 'Settlement (বুঝিয়ে নেওয়া)',
   189→      category: 'Settlement',
   190→    })
   191→    toast.success('Settled successfully')
   192→    setShowSettle(false)
   193→    refetch()
   194→  }
   195→
   196→  return (
   197→    <motion.div
   198→      initial={{ opacity: 0, x: 20 }}
   199→      animate={{ opacity: 1, x: 0 }}
   200→      className="space-y-4"
   201→    >
   202→      {/* PRD Part 32 §3.1: Defaulter alert banner */}
   203→      {activeDefaulter && (
   204→        <DefaulterAlertBanner
   205→          amount={activeDefaulter.defaultAmount}
   206→          merchantName={activeDefaulter.merchantName}
   207→          partyName={data.name}
   208→        />
   209→      )}
   210→
   211→      {/* Header */}
   212→      <div className="flex items-center gap-2">
   213→        <button
   214→          onClick={handleBack}
   215→          className="w-10 h-10 rounded-full hover:bg-muted flex items-center justify-center"
   216→          aria-label="Back"
   217→        >
   218→          <ArrowLeft className="w-5 h-5" />
   219→        </button>
   220→        <h2 className="text-base font-semibold flex-1 truncate">{data.name}</h2>
   221→        <button
   222→          onClick={() => setEditingPartyId(partyId)}
   223→          className="w-10 h-10 rounded-full hover:bg-muted flex items-center justify-center"
   224→          aria-label="Edit"
   225→        >
   226→          <FileEdit className="w-4 h-4" />
   227→        </button>
   228→      </div>
   229→
   230→      {/* Profile card */}
   231→      <div className="rounded-2xl bg-card border border-border p-5 shadow-sm">
   232→        <div className="flex items-start gap-3">
   233→          <div className="w-14 h-14 rounded-2xl bg-gradient-to-br from-emerald-400 to-emerald-600 flex items-center justify-center text-white text-xl font-bold shrink-0">
   234→            {data.name.charAt(0).toUpperCase()}
   235→          </div>
   236→          <div className="flex-1 min-w-0">
   237→            <div className="flex items-center gap-2 flex-wrap">
   238→              <h3 className="text-base font-semibold truncate">{data.name}</h3>
   239→              <span className={`text-[10px] font-bold px-2 py-0.5 rounded-full ${meta.bg} ${meta.color}`}>
   240→                {data.qualityGrade} · {meta.desc}
   241→              </span>
   242→            </div>
   243→            <p className="text-xs text-muted-foreground capitalize">{t(`common.${data.type}`)}</p>
   244→            {data.phone && (
   245→              <a href={`tel:${data.phone}`} className="text-xs text-primary flex items-center gap-1 mt-1">
   246→                <Phone className="w-3 h-3" /> {data.phone}
   247→              </a>
   248→            )}
   249→          </div>
   250→        </div>
   251→
   252→        {/* Balance */}
   253→        <div className="mt-4 p-4 rounded-xl bg-muted/50">
   254→          <p className="text-xs text-muted-foreground mb-0.5">
   255→            {isReceivable ? t('khata.outstanding') : isPayable ? 'আপনি দেবেন' : 'ব্যালেন্স'}
   256→          </p>
   257→          <p className={`text-2xl font-bold tabular ${isReceivable ? 'text-emerald-600' : isPayable ? 'text-red-600' : 'text-foreground'}`}>
   258→            {formatCurrency(Math.abs(data.balance), currency)}
   259→          </p>
   260→          {data.creditLimit && (
   261→            <p className="text-[11px] text-muted-foreground mt-1">
   262→              Credit Limit: {formatCurrency(data.creditLimit, currency)}
   263→            </p>
   264→          )}
   265→        </div>
   266→
   267→        {/* Action buttons */}
   268→        <div className="grid grid-cols-4 gap-2 mt-4">
   269→          {data.phone && (
   270→            <Button
   271→              variant="outline"
   272→              className="flex flex-col items-center gap-1 h-auto py-2.5"
   273→              onClick={() => window.location.href = `tel:${data.phone}`}
   274→            >
   275→              <Phone className="w-4 h-4" />
   276→              <span className="text-[10px]">{t('khata.call')}</span>
   277→            </Button>
   278→          )}
   279→          <Button
   280→            variant="outline"
   281→            className="flex flex-col items-center gap-1 h-auto py-2.5"
   282→            onClick={() => setShowTxn(true)}
   283→          >
   284→            <Plus className="w-4 h-4" />
   285→            <span className="text-[10px]">Entry</span>
   286→          </Button>
   287→          <Button
   288→            variant="outline"
   289→            className="flex flex-col items-center gap-1 h-auto py-2.5"
   290→            onClick={() => {
   291→              useAppStore.getState().setPendingNewCustomer(data.id, data.name)
   292→              setReturnToView('khata')
   293→              setActiveView('sale-pad')
   294→            }}
   295→          >
   296→            <Zap className="w-4 h-4" />
   297→            <span className="text-[10px]">Quick Sale</span>
   298→          </Button>
   299→          <Button
   300→            variant="outline"
   301→            className="flex flex-col items-center gap-1 h-auto py-2.5"
   302→            onClick={() => setShowSettle(true)}
   303→          >
   304→            <CheckCircle2 className="w-4 h-4" />
   305→            <span className="text-[10px]">{t('khata.settleUp')}</span>
   306→          </Button>
   307→        </div>
   308→
   309→        {/* PRD Part 32 §4: Biometric linking buttons */}
   310→        <div className="grid grid-cols-3 gap-2 mt-2">
   311→          <Button
   312→            variant="outline"
   313→            className="flex items-center justify-center gap-1.5 h-9 text-[11px]"
   314→            onClick={() => setShowFingerprintRegister(true)}
   315→          >
   316→            <Fingerprint className="w-3.5 h-3.5 text-emerald-600" /> Fingerprint
   317→          </Button>
   318→          {isCustomer && (
   319→            <Button
   320→              variant="outline"
   321→              className="flex items-center justify-center gap-1.5 h-9 text-[11px]"
   322→              onClick={() => setShowFamily(true)}
   323→            >
   324→              <Users className="w-3.5 h-3.5 text-cyan-600" /> Family
   325→            </Button>
   326→          )}
   327→          {isSupplier && (
   328→            <Button
   329→              variant="outline"
   330→              className="flex items-center justify-center gap-1.5 h-9 text-[11px]"
   331→              onClick={() => setShowPartner(true)}
   332→            >
   333→              <Briefcase className="w-3.5 h-3.5 text-purple-600" /> Partner
   334→            </Button>
   335→          )}
   336→        </div>
   337→      </div>
   338→
   339→      {/* PRD Part 32 §3.2: AI Credit Trust Score (customers only) */}
   340→      {isCustomer && (
   341→        <TrustScoreCard partyId={partyId} partyName={data.name} />
   342→      )}
   343→
   344→      {/* Transactions — multi-select + share (PRD Part 6 §2 + Part 7 §4) */}
   345→      <div className="rounded-2xl bg-card border border-border p-4 shadow-sm">
   346→        <div className="flex items-center justify-between mb-3">
   347→          <h3 className="text-sm font-semibold">{t('khata.transactions')}</h3>
   348→          <div className="flex items-center gap-2">
   349→            {data.transactions.length > 0 && !multiSelectMode && (
   350→              <button
   351→                onClick={handleShareStatement}
   352→                className="text-[10px] font-medium text-primary bg-primary/10 px-2 py-1 rounded-lg flex items-center gap-1"
   353→              >
   354→                <FileText className="w-3 h-3" /> Share Statement
   355→              </button>
   356→            )}
   357→            <span className="text-xs text-muted-foreground">{data.transactions.length}</span>
   358→          </div>
   359→        </div>
   360→
   361→        {/* Multi-select control bar (PRD Part 7 §4.3) */}
   362→        {multiSelectMode && (
   363→          <div className="flex items-center justify-between gap-2 mb-3 p-2 rounded-xl bg-primary/5 border border-primary/20">
   364→            <div className="flex items-center gap-2">
   365→              <button onClick={selectAllTx} className="text-[10px] font-medium text-primary px-2 py-1 rounded-lg bg-primary/10">Select All</button>
   366→              <button onClick={deselectAllTx} className="text-[10px] font-medium text-muted-foreground px-2 py-1 rounded-lg bg-muted">Deselect All</button>
   367→            </div>
   368→            <span className="text-[10px] text-muted-foreground">{selectedTxIds.size} selected</span>
   369→          </div>
   370→        )}
   371→
   372→        {data.transactions.length === 0 ? (
   373→          <p className="text-sm text-muted-foreground py-6 text-center">No transactions yet</p>
   374→        ) : (
   375→          <>
   376→            <div className="space-y-1 max-h-96 overflow-y-auto scroll-area">
   377→              {data.transactions.map((tx) => {
   378→                const isCredit = tx.type === 'credit'
   379→                const isSelected = selectedTxIds.has(tx.id)
   380→                return (
   381→                  <TxRow
   382→                    key={tx.id}
   383→                    tx={tx}
   384→                    isCredit={isCredit}
   385→                    isSelected={isSelected}
   386→                    multiSelectMode={multiSelectMode}
   387→                    currency={currency}
   388→                    onLongPress={() => { setMultiSelectMode(true); setSelectedTxIds(new Set([tx.id])) }}
   389→                    onToggle={() => toggleTxSelection(tx.id)}
   390→                  />
   391→                )
   392→              })}
   393→            </div>
   394→            {/* Share Selected button (PRD Part 7 §4.4) */}
   395→            {multiSelectMode && selectedTxIds.size > 0 && (
   396→              <button
   397→                onClick={handleShareSelected}
   398→                className="w-full mt-3 h-11 rounded-xl bg-primary text-primary-foreground font-medium text-sm flex items-center justify-center gap-2"
   399→              >
   400→                <Share2 className="w-4 h-4" /> Share Selected ({selectedTxIds.size})
   401→              </button>
   402→            )}
   403→          </>
   404→        )}
   405→      </div>
   406→
   407→      {/* Invoices */}
   408→      {data.invoices.length > 0 && (
   409→        <div className="rounded-2xl bg-card border border-border p-4 shadow-sm">
   410→          <h3 className="text-sm font-semibold mb-3">Invoices ({data.invoices.length})</h3>
   411→          <div className="space-y-1 max-h-64 overflow-y-auto scroll-area">
   412→            {data.invoices.map((inv) => (
   413→              <button
   414→                key={inv.id}
   415→                onClick={() => { setSelectedInvoiceId(inv.id); setActiveView('billing') }}
   416→                className="w-full flex items-center gap-3 p-2 rounded-lg hover:bg-muted/50 text-left"
   417→              >
   418→                <Receipt className="w-4 h-4 text-muted-foreground shrink-0" />
   419→                <div className="flex-1 min-w-0">
   420→                  <p className="text-sm font-medium truncate">{inv.invoiceNumber}</p>
   421→                  <p className="text-[11px] text-muted-foreground">{formatDate(inv.createdAt)}</p>
   422→                </div>
   423→                <div className="text-right">
   424→                  <p className="text-sm font-semibold tabular">{formatCurrency(inv.grandTotal, currency)}</p>
   425→                  <span className={`text-[10px] ${inv.status === 'paid' ? 'text-emerald-600' : inv.status === 'partial' ? 'text-amber-600' : 'text-red-600'}`}>
   426→                    {inv.status}
   427→                  </span>
   428→                </div>
   429→              </button>
   430→            ))}
   431→          </div>
   432→        </div>
   433→      )}
   434→
   435→      {/* PRD Part 24: Source Products section for suppliers */}
   436→      {isSupplier && supplierCatalog && supplierCatalog.length > 0 && (
   437→        <div className="rounded-2xl bg-card border border-border p-4 shadow-sm">
   438→          <h3 className="text-sm font-semibold mb-3 flex items-center gap-1.5">
   439→            <Package className="w-4 h-4 text-purple-600" /> Source Products ({supplierCatalog.length})
   440→          </h3>
   441→          <div className="space-y-2 max-h-72 overflow-y-auto scroll-area">
   442→            {supplierCatalog.map((item) => (
   443→              <div key={item.id} className="flex items-center gap-2 p-2 rounded-lg bg-muted/50">
   444→                <div className="flex-1 min-w-0">
   445→                  <p className="text-xs font-medium truncate">{item.productName}</p>
   446→                  <p className="text-[10px] text-muted-foreground">
   447→                    Base: {formatCurrency(item.basePrice, currency)} · Landed: {formatCurrency(item.perUnitLandedCost || 0, currency)}
   448→                  </p>
   449→                </div>
   450→                <button
   451→                  onClick={() => {
   452→                    setCompareProduct(item.productName)
   453→                    setShowCompare(true)
   454→                  }}
   455→                  className="text-[10px] font-medium text-purple-600 bg-purple-100 dark:bg-purple-900/30 px-2 py-1 rounded-lg flex items-center gap-0.5 shrink-0"
   456→                >
   457→                  <Award className="w-3 h-3" /> Compare
   458→                </button>
   459→              </div>
   460→            ))}
   461→          </div>
   462→        </div>
   463→      )}
   464→
   465→      <TransactionForm open={showTxn} onOpenChange={setShowTxn} party={data} />
   466→      <SettleUpDialog open={showSettle} onOpenChange={setShowSettle} party={data} onConfirm={handleSettle} />
   467→      <PartyForm
   468→        open={!!editingPartyId}
   469→        onOpenChange={(o) => { if (!o) setEditingPartyId(null) }}
   470→        partyId={editingPartyId}
   471→      />
   472→      <ShareSheet
   473→        open={showShareSheet}
   474→        onClose={() => setShowShareSheet(false)}
   475→        customerName={data.name}
   476→        customerPhone={data.phone ?? null}
   477→        shareText={shareSheetText}

   513→function FingerprintRegisterDialog({
   514→  open, onOpenChange, partyId, partyName, onDone,
   515→}: {
   516→  open: boolean
   517→  onOpenChange: (o: boolean) => void
   518→  partyId: string
   519→  partyName: string
   520→  onDone: () => void
   521→}) {
   522→  const [scanning, setScanning] = useState(false)
   523→  const [hand, setHand] = useState('right')
   524→  const [finger, setFinger] = useState('thumb')

   528→    setScanning(true)

   532→      await apiPost('/api/fingerprints', {
   533→        partyId,
   534→        role: 'primary',
   535→        hand,
   536→        finger,
   537→        scannerType,
   538→      })
   539→      toast.success(`Fingerprint registered for ${partyName}`, {
   540→        description: `${hand} ${finger} · ${scannerType === 'external' ? 'USB OTG (MFS100)' : 'Native sensor'}`,
   541→      })

   543→      onDone()
   544→    } catch (e: any) {
   545→      toast.error(e.message || 'Failed to register fingerprint')
   546→    } finally {
   547→      setScanning(false)
   548→    }
   549→  }
   550→
   551→  return (
   552→    <Dialog open={open} onOpenChange={onOpenChange}>
   553→      <DialogContent className="max-w-md">
   554→        <DialogHeader>
   555→          <DialogTitle className="flex items-center gap-2">

   571→              >
   572→                <option value="right">Right</option>
   573→                <option value="left">Left</option>
   574→              </select>

   582→              >
   583→                <option value="thumb">Thumb</option>
   584→                <option value="index">Index</option>

   589→            </div>

   591→          <div className="space-y-1.5">
   592→            <Label className="text-xs">Scanner Type</Label>
   593→            <div className="grid grid-cols-2 gap-2">
   594→              <button
   595→                onClick={() => setScannerType('native')}

   597→                  scannerType === 'native' ? 'border-emerald-500 bg-emerald-500/10 text-emerald-600' : 'border-border bg-background'

   683→// Transaction row with long-press multi-select (PRD Part 7 §4)
   684→function TxRow({ tx, isCredit, isSelected, multiSelectMode, currency, onLongPress, onToggle }: {
   685→  tx: any
   686→  isCredit: boolean
   687→  isSelected: boolean
   688→  multiSelectMode: boolean
   689→  currency: string
   690→  onLongPress: () => void
   691→  onToggle: () => void
   692→}) {
   693→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   694→  const longPressedRef = useRef(false)
   695→
   696→  const handleStart = () => {
   697→    longPressedRef.current = false
   698→    timerRef.current = setTimeout(() => {
   699→      longPressedRef.current = true
   700→      onLongPress()
   701→    }, 500)
   702→  }
   703→  const handleEnd = () => {
   704→    if (timerRef.current) { clearTimeout(timerRef.current); timerRef.current = null }
   705→    if (!longPressedRef.current && multiSelectMode) {
   706→      onToggle()
   707→    }
   708→  }
   709→
   710→  return (
   711→    <div
   712→      onTouchStart={handleStart}
   713→      onTouchEnd={handleEnd}
   714→      onMouseDown={handleStart}
   715→      onMouseUp={handleEnd}
   716→      onMouseLeave={() => { if (timerRef.current) { clearTimeout(timerRef.current); timerRef.current = null } }}
   717→      className={`flex items-center gap-3 p-2 rounded-lg transition-colors cursor-pointer select-none ${
   718→        isSelected ? 'bg-primary/10 border border-primary/30' : 'hover:bg-muted/50'
   719→      }`}
   720→    >
   721→      {multiSelectMode && (
   722→        <span className={`w-5 h-5 rounded-md border-2 shrink-0 flex items-center justify-center ${
   723→          isSelected ? 'bg-primary border-primary text-primary-foreground' : 'border-muted-foreground'
   724→        }`}>
   725→          {isSelected && <CheckCircle2 className="w-3 h-3" />}
   726→        </span>
   727→      )}
   728→      <span className={`w-8 h-8 rounded-full flex items-center justify-center shrink-0 ${isCredit ? 'bg-emerald-100 dark:bg-emerald-900/30' : 'bg-red-100 dark:bg-red-900/30'}`}>
   729→        {isCredit ? <ArrowDownLeft className="w-4 h-4 text-emerald-600" /> : <ArrowUpRight className="w-4 h-4 text-red-600" />}
   730→      </span>
   731→      <div className="flex-1 min-w-0">
   732→        <p className="text-sm font-medium truncate">{tx.description || tx.type}</p>
   733→        <p className="text-[11px] text-muted-foreground">{formatDate(tx.createdAt)}</p>
   734→      </div>
   735→      <span className={`text-sm font-semibold tabular shrink-0 ${isCredit ? 'text-emerald-600' : 'text-red-600'}`}>
   736→        {isCredit ? '+' : '-'}{formatCurrency(tx.amount, currency)}
   737→      </span>
   738→    </div>
   739→  )
   740→}
